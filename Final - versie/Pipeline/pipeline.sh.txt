#!/bin/bash

ZONE="europe-west1-b"
HOME_DIR="/home/larissa_mbonazolusakueno"
PIPELINE_START=$(date +%s)

ALL_VMS="dc1-cassandra dc1-kafka dc1-spark dc2-cassandra dc2-kafka dc2-spark dc3-cassandra dc3-kafka dc3-spark dc4-cassandra dc4-kafka dc4-spark"
KAFKA_VMS="dc1-kafka dc2-kafka dc3-kafka dc4-kafka"
SPARK_VMS="dc1-spark dc2-spark dc3-spark dc4-spark"
CASSANDRA_VMS="dc1-cassandra dc2-cassandra dc3-cassandra dc4-cassandra"

echo "=========================================================="
echo "   EUROVISION AUDIT SYSTEEM - VOLLEDIGE PIPELINE"
echo "   Odisee Hogeschool - Mega Data Challenge 2025-2026"
echo "   Gestart op: $(date '+%Y-%m-%d %H:%M:%S')"
echo "=========================================================="
echo ""

# STAP 1 — START ALLE 12 VM'S OP
echo "STAP 1 - Alle 12 VM's opstarten..."
echo "----------------------------------------------------------"
gcloud compute instances start $ALL_VMS --zone=$ZONE --quiet
if [ $? -eq 0 ]; then
    echo "✓ Alle 12 VM's opgestart"
else
    echo "✗ Fout bij opstarten VM's"
    exit 1
fi
echo ""

# STAP 2 — WACHT TOT ALLE VM'S SSH-BEREIKBAAR ZIJN
echo "STAP 2 - Wachten tot alle VM's SSH-bereikbaar zijn..."
echo "----------------------------------------------------------"
for VM in $ALL_VMS; do
    echo -n "  Wachten op $VM... "
    BEREIKBAAR=0
    for i in $(seq 1 30); do
        gcloud compute ssh $VM --zone=$ZONE --command="echo ok" --quiet 2>/dev/null
        if [ $? -eq 0 ]; then
            BEREIKBAAR=1
            break
        fi
        sleep 5
    done
    if [ $BEREIKBAAR -eq 1 ]; then
        echo "✓ bereikbaar"
    else
        echo "✗ NIET bereikbaar na 150s"
    fi
done
echo ""

# STAP 3 — CHECK + START DOCKER CONTAINERS
echo "STAP 3 - Docker containers controleren en starten..."
echo "----------------------------------------------------------"

echo "  [Kafka containers]"
for VM in $KAFKA_VMS; do
    gcloud compute ssh $VM --zone=$ZONE --quiet 2>/dev/null --command="
        KAFKA=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-kafka-1 2>/dev/null)
        ZOO=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-zookeeper-1 2>/dev/null)
        if [ \"\$ZOO\" != 'true' ]; then
            sudo docker start larissa_mbonazolusakueno-zookeeper-1
            sleep 10
        fi
        if [ \"\$KAFKA\" != 'true' ]; then
            sudo docker start larissa_mbonazolusakueno-kafka-1
            sleep 5
        fi
        KAFKA2=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-kafka-1 2>/dev/null)
        ZOO2=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-zookeeper-1 2>/dev/null)
        echo '  $VM: Zookeeper='\$ZOO2' | Kafka='\$KAFKA2
    "
done

echo "  [Spark containers]"
for VM in $SPARK_VMS; do
    gcloud compute ssh $VM --zone=$ZONE --quiet 2>/dev/null --command="
        MASTER=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-spark-master-1 2>/dev/null)
        WORKER=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-spark-worker-1 2>/dev/null)
        if [ \"\$MASTER\" != 'true' ]; then
            sudo docker start larissa_mbonazolusakueno-spark-master-1
            sleep 5
        fi
        if [ \"\$WORKER\" != 'true' ]; then
            sudo docker start larissa_mbonazolusakueno-spark-worker-1
            sleep 5
        fi
        MASTER2=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-spark-master-1 2>/dev/null)
        WORKER2=\$(sudo docker inspect -f '{{.State.Running}}' larissa_mbonazolusakueno-spark-worker-1 2>/dev/null)
        echo '  $VM: Spark-master='\$MASTER2' | Spark-worker='\$WORKER2
    "
done

echo "  [Cassandra + Hadoop containers]"
for VM in $CASSANDRA_VMS; do
    gcloud compute ssh $VM --zone=$ZONE --quiet 2>/dev/null --command="
        CASS=\$(sudo docker inspect -f '{{.State.Running}}' eurovision-cassandra 2>/dev/null)
        NAME=\$(sudo docker inspect -f '{{.State.Running}}' eurovision-namenode 2>/dev/null)
        DATA=\$(sudo docker inspect -f '{{.State.Running}}' eurovision-datanode 2>/dev/null)
        if [ \"\$NAME\" != 'true' ]; then
            sudo docker start eurovision-namenode
            sleep 10
        fi
        if [ \"\$DATA\" != 'true' ]; then
            sudo docker start eurovision-datanode
            sleep 5
        fi
        if [ \"\$CASS\" != 'true' ]; then
            sudo docker start eurovision-cassandra
            sleep 15
        fi
        CASS2=\$(sudo docker inspect -f '{{.State.Running}}' eurovision-cassandra 2>/dev/null)
        NAME2=\$(sudo docker inspect -f '{{.State.Running}}' eurovision-namenode 2>/dev/null)
        DATA2=\$(sudo docker inspect -f '{{.State.Running}}' eurovision-datanode 2>/dev/null)
        echo '  $VM: Cassandra='\$CASS2' | Namenode='\$NAME2' | Datanode='\$DATA2
    "
done
echo ""

# STAP 4 — TOON ACTIEVE CONTAINERS PER VM
echo "STAP 4 - Overzicht actieve containers per VM:"
echo "----------------------------------------------------------"
for VM in $ALL_VMS; do
    echo "  [$VM]"
    gcloud compute ssh $VM --zone=$ZONE --quiet 2>/dev/null --command="
        sudo docker ps --format '    {{.Names}} | {{.Status}} | {{.Image}}'
    "
done
echo ""

# STAP 5 — START ALLE 12 PRODUCERS PARALLEL
echo "STAP 5 - Alle 12 producers starten (stemmen genereren)..."
echo "----------------------------------------------------------"
for VM in $ALL_VMS; do
    gcloud compute ssh $VM --zone=$ZONE --quiet 2>/dev/null --command="nohup python3 $HOME_DIR/producer.py > $HOME_DIR/producer_output.txt 2>&1 &" &
done
wait
echo "✓ Alle 12 VM's sturen stemmen naar Kafka"
echo ""

# STAP 6 — WACHT 30 SECONDEN
echo "STAP 6 - Wachten 30 seconden voor stemmen binnenkomen..."
echo "----------------------------------------------------------"
for i in $(seq 30 -1 1); do
    echo -ne "  Nog $i seconden...\r"
    sleep 1
done
echo "✓ Wachttijd voorbij                    "
echo ""

# STAP 7 — THROUGHPUT RAPPORT
echo "STAP 7 - Throughput per VM:"
echo "----------------------------------------------------------"
for VM in $ALL_VMS; do
    RESULT=$(gcloud compute ssh $VM --zone=$ZONE --quiet 2>/dev/null --command="grep 'Gemiddelde throughput' $HOME_DIR/producer_output.txt 2>/dev/null | tail -1")
    if [ -z "$RESULT" ]; then
        RESULT="(nog bezig of geen output)"
    fi
    echo "  $VM: $RESULT"
done
echo ""

# STAP 8 — DATA INTEGRITEIT (NFR-03)
echo "STAP 8 - Data Integriteit Test (NFR-03)..."
echo "----------------------------------------------------------"
gcloud compute ssh dc1-cassandra --zone=$ZONE --quiet 2>/dev/null --command="cd $HOME_DIR && python3 demo_integriteit.py 2>&1 | tee demo_integriteit_output.txt"
echo ""

# STAP 9 — CONSISTENCY CHECK (NFR-04)
echo "STAP 9 - Consistency Check (NFR-04)..."
echo "----------------------------------------------------------"
gcloud compute ssh dc1-cassandra --zone=$ZONE --quiet 2>/dev/null --command="cd $HOME_DIR && python3 demo_consistency.py 2>&1 | tee demo_consistency_output.txt"
echo ""

# STAP 10 — EINDRAPPORT
PIPELINE_END=$(date +%s)
PIPELINE_DUUR=$((PIPELINE_END - PIPELINE_START))
MINUTEN=$((PIPELINE_DUUR / 60))
SECONDEN=$((PIPELINE_DUUR % 60))

echo "=========================================================="
echo "   EINDRAPPORT EUROVISION AUDIT PIPELINE"
echo "=========================================================="
echo "  Stap 1:  VM's opgestart             ✓"
echo "  Stap 2:  SSH bereikbaarheid         ✓"
echo "  Stap 3:  Docker containers gestart  ✓"
echo "  Stap 4:  Containers overzicht       ✓"
echo "  Stap 5:  Producers gestart          ✓"
echo "  Stap 6:  Stemmen gegenereerd        ✓"
echo "  Stap 7:  Throughput gemeten         ✓"
echo "  Stap 8:  NFR-03 integriteit         ✓"
echo "  Stap 9:  NFR-04 consistency         ✓"
echo ""
echo "  VM's blijven draaien na pipeline!"
echo ""
echo "  Gestart  : $(date -d @$PIPELINE_START '+%Y-%m-%d %H:%M:%S')"
echo "  Gestopt  : $(date '+%Y-%m-%d %H:%M:%S')"
echo "  Totale uitvoeringstijd: $MINUTEN min $SECONDEN sec"
echo "=========================================================="

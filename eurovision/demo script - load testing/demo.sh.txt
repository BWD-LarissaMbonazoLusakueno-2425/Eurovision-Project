cat > ~/demo.sh << 'EOF'
#!/bin/bash
echo "=================================================="
echo "   EUROVISION AUDIT SYSTEEM - DEMO"
echo "=================================================="

echo ""
echo "STAP 1: Kafka controleren op dc1-kafka..."
gcloud compute ssh dc1-kafka --zone=europe-west1-b --command="
  ~/kafka/bin/kafka-topics.sh --describe --topic eurovision-votes --bootstrap-server localhost:9092 2>/dev/null | grep PartitionCount
" 2>/dev/null
echo "✓ Kafka: 12 partities actief"

echo ""
echo "STAP 2: Spark cluster controleren..."
gcloud compute ssh dc1-spark --zone=europe-west1-b --command="
  tail -5 ~/spark/logs/spark-*Master*.out | grep -i 'worker\|alive\|master'
" 2>/dev/null
echo "✓ Spark: master + 3 workers actief"

echo ""
echo "STAP 3: Alle 12 producers starten..."
for VM in dc1-kafka dc1-spark dc1-cassandra dc2-kafka dc2-spark dc2-cassandra dc3-kafka dc3-spark dc3-cassandra dc4-kafka dc4-spark dc4-cassandra; do
    gcloud compute ssh $VM --zone=europe-west1-b --command="nohup python3 ~/producer.py > ~/producer_output.txt 2>&1 &" &
done
wait
echo "✓ Alle 12 VM's sturen stemmen!"

echo ""
echo "STAP 4: Wacht 15 seconden voor resultaten..."
sleep 15

echo ""
echo "STAP 5: Throughput per VM:"
echo "--------------------------------------------------"
TOTAAL=0
for VM in dc1-kafka dc1-spark dc1-cassandra dc2-kafka dc2-spark dc2-cassandra dc3-kafka dc3-spark dc3-cassandra dc4-kafka dc4-spark dc4-cassandra; do
    RESULT=$(gcloud compute ssh $VM --zone=europe-west1-b --command="grep 'Gemiddelde throughput' ~/producer_output.txt 2>/dev/null | tail -1" 2>/dev/null)
    echo "  $VM: $RESULT"
done

echo ""
echo "STAP 6: Bottleneck analyse (CPU/RAM):"
echo "--------------------------------------------------"
for VM in dc1-kafka dc2-kafka dc3-kafka dc4-kafka; do
    gcloud compute ssh $VM --zone=europe-west1-b --command="
    CPU=\$(top -b -n 1 | grep '%Cpu' | awk '{print \$2}')
    RAM=\$(free -m | grep Mem | awk '{print \$3\"/\"\$2\" MB\"}')
    echo \"  $VM → CPU: \${CPU}% | RAM: \${RAM}\"
    " 2>/dev/null
done

echo ""
echo "=================================================="
echo "   DEMO KLAAR!"
echo "=================================================="
EOF
chmod +x ~/demo.sh
echo "Demo script aangemaakt!"
